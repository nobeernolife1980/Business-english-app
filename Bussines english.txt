    for(let i=30; i<500; i++) {
        questions.push({...questions[i%30], grammar: `Level ${Math.floor(i/10)+1}`});
    }

    let currentIndex = 0, recognition, mediaRecorder, audioChunks = [], audioUrl, synth = window.speechSynthesis, stream;

    async function setupAudio() {
        if (!stream) {
            try {
                stream = await navigator.mediaDevices.getUserMedia({ audio: true });
                mediaRecorder = new MediaRecorder(stream);
                mediaRecorder.ondataavailable = e => audioChunks.push(e.data);
                mediaRecorder.onstop = () => { audioUrl = URL.createObjectURL(new Blob(audioChunks, { type: 'audio/wav' })); };
            } catch (e) { alert("Microphone access is required."); return; }
        }
        if (!recognition) {
            const SpeechRec = window.SpeechRecognition || window.webkitSpeechRecognition;
            recognition = new SpeechRec();
            recognition.lang = 'en-US';
            recognition.onresult = (e) => {
                const speech = e.results[0][0].transcript;
                if(mediaRecorder.state === "recording") mediaRecorder.stop();
                showResult(speech);
            };
            recognition.onerror = () => { recordBtn.innerText = "🎤 Error (Retry)"; recordBtn.className = "btn-primary"; };
        }
    }

    const recordBtn = document.getElementById('record-btn');
    const resultArea = document.getElementById('result-area');

    function init() {
        const q = questions[currentIndex];
        document.getElementById('grammar-label').innerText = q.grammar;
        document.getElementById('jp-text').innerText = q.jp;
        document.getElementById('progress').innerText = `${currentIndex + 1} / 500`;
        resultArea.classList.add('hidden');
        document.getElementById('next-btn').classList.add('hidden');
        document.getElementById('retry-btn').classList.add('hidden');
        recordBtn.classList.remove('hidden');
        recordBtn.innerText = "🎤 Click to Speak";
        recordBtn.className = "btn-primary";
    }

    recordBtn.onclick = async () => {
        await setupAudio();
        if(!stream) return;
        audioChunks = [];
        recognition.start();
        mediaRecorder.start();
        recordBtn.innerText = "Listening...";
        recordBtn.className = "btn-recording";
    };

    function showResult(userSpeech) {
        const q = questions[currentIndex];
        document.getElementById('user-speech').innerText = `"${userSpeech}"`;
        document.getElementById('model-answer').innerHTML = q.display.split(' / ').map(p => p + `<span class="slash">/</span>`).join('');
        const isPerfect = userSpeech.toLowerCase().replace(/[.\s]/g,'') === q.en.toLowerCase().replace(/[.\s]/g,'');
        document.getElementById('feedback-en').innerText = isPerfect ? "Perfect! Well done." : "Good try!";
        document.getElementById('feedback-jp').innerText = isPerfect ? "完璧な正解です！" : "惜しい！模範解答を聞いて再挑戦しましょう。";
        document.getElementById('grammar-point-text').innerText = q.point;
        document.getElementById('alt-text').innerText = q.alt;
        resultArea.classList.remove('hidden');
        recordBtn.classList.add('hidden');
        document.getElementById('next-btn').classList.remove('hidden');
        document.getElementById('retry-btn').classList.remove('hidden');
        playModel(1.0);
    }

    window.playModel = (speed) => {
        synth.cancel();
        const utter = new SpeechSynthesisUtterance(questions[currentIndex].en);
        utter.lang = 'en-US'; utter.rate = speed;
        synth.speak(utter);
    };

    document.getElementById('replay-btn').onclick = () => { if(audioUrl) new Audio(audioUrl).play(); };
    document.getElementById('next-btn').onclick = () => { currentIndex = (currentIndex + 1) % 500; init(); };
    document.getElementById('retry-btn').onclick = init;
    init();
